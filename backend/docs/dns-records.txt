; ─── DNS Configuration Records for Domain: kalceria.com ───────────────────────
; Dual-Channel Email Delivery:
; Channel 1: Auth & OTP (auth.kalceria.com)
; Channel 2: Notifications & Broadcasts (notify.kalceria.com / kalceria.com)

; ==============================================================================
; 1. RESEND DOMAIN 1: AUTH & OTP (auth.kalceria.com)
; ==============================================================================
; MX Record for Inbound Feedback / Bounce Handling
auth.kalceria.com. IN MX 10 feedback-smtp.us-east-1.amazonses.com.

; SPF Record (Authorize Resend SES mail servers)
auth.kalceria.com. IN TXT "v=spf1 include:amazonses.com ~all"

; DKIM Records (Generated by Resend in dashboard.resend.com/domains)
resend._domainkey.auth.kalceria.com. IN TXT "k=rsa; p=<PASTE_RESEND_AUTH_DKIM_KEY_HERE>"

; DMARC Policy for auth subdomain
_dmarc.auth.kalceria.com. IN TXT "v=DMARC1; p=reject; pct=100; rua=mailto:dmarc@kalceria.com"


; ==============================================================================
; 2. RESEND DOMAIN 2: NOTIFICATIONS (notify.kalceria.com)
; ==============================================================================
; MX Record
notify.kalceria.com. IN MX 10 feedback-smtp.us-east-1.amazonses.com.

; SPF Record
notify.kalceria.com. IN TXT "v=spf1 include:amazonses.com ~all"

; DKIM Records (Generated by Resend for notify subdomain)
resend._domainkey.notify.kalceria.com. IN TXT "k=rsa; p=<PASTE_RESEND_NOTIF_DKIM_KEY_HERE>"

; DMARC Policy for notify subdomain
_dmarc.notify.kalceria.com. IN TXT "v=DMARC1; p=quarantine; pct=100; rua=mailto:dmarc@kalceria.com"


; ==============================================================================
; 3. ROOT DOMAIN UNIFIED FALLBACK (kalceria.com)
; ==============================================================================
; SPF (Combined Google Workspace + Resend)
@ IN TXT "v=spf1 include:_spf.google.com include:amazonses.com ~all"

; Google Workspace DKIM
google._domainkey IN TXT "v=DKIM1; k=rsa; p=<PASTE_GOOGLE_DKIM_PUBLIC_KEY_HERE>"

; Root DMARC Policy
_dmarc IN TXT "v=DMARC1; p=reject; rua=mailto:dmarc-reports@kalceria.com; fo=1; adkim=s; aspf=s"
